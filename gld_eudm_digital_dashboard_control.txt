spark.sql(f"""
CREATE OR REPLACE TEMPORARY VIEW sf_sales_fact AS
SELECT * FROM (
    SELECT *,
    row_number() OVER (
        PARTITION BY 
            datasource_num_id, 
            invoice_no, 
            order_number, 
            branch_id, 
            COALESCE(CAST(invoice_line_no AS STRING), concat(CAST(sales_amount AS STRING), CAST(quantity AS STRING)))
        ORDER BY updated_ts DESC
    ) AS row_num
    FROM (
        SELECT
            datasource_num_id, 
            TRIM(branch_id) AS branch_id, 
            to_date(date_invoice) AS date_invoice, 
            sales_amount, 
            currency, 
            customer_dim_key,
            TRIM(invoice_no) AS invoice_no, 
            TRIM(wesnet_order_no) AS wesnet_order_no, 
            TRIM(order_number) AS order_number, 
            order_integration_id, 
            accounting_include_flag,
            sales_transfer_flag, 
            kit_component_flag, 
            TRIM(sim_description) AS sim_description, 
            TRIM(inventory_org_code) AS inventory_org_code,
            cost_amount, 
            quantity, 
            market_cost, 
            TRIM(transaction_type) AS transaction_type, 
            invoice_line_no, 
            conversion_rate,
            budget_conversion_rate, 
            TRIM(flex_1) AS flex_1, 
            dl_load_date, 
            updated_ts
        FROM brz_teradata.udm_sales_fact 
        WHERE dl_load_date > '{min_cdc_value}'
          AND datasource_num_id IN (3000, 4000, 5000, 6000, 7000, 8000, 9000, 10000, 11000, 12001, 12002, 12003, 12004, 12005, 13000, 14000)
    )
)
WHERE row_num = 1
""")


# COMMAND ----------

spark.sql(f"""
CREATE OR REPLACE TEMPORARY VIEW of_order_fact AS
SELECT * FROM (
    SELECT *,
    row_number() OVER (
        PARTITION BY datasource_num_id, sales_order_no, branch_id, order_integration_id, flex_1, ent_type, invoice_no
        ORDER BY order_updated_ts DESC
    ) AS row_num
    FROM (
        SELECT
            datasource_num_id, 
            TRIM(sales_order_no) AS sales_order_no, 
            TRIM(branch_id) AS branch_id, 
            order_integration_id, 
            TRIM(flex_1) AS flex_1,
            TRIM(ent_type) AS ent_type, 
            TRIM(invoice_no) AS invoice_no, 
            to_date(date_invoice) AS date_invoice, 
            WesnetLine, 
            dl_load_date, 
            order_updated_ts,
            TRIM(customer_po_no) AS customer_po_no
        FROM brz_teradata.udm_order_fact
        WHERE dl_load_date > '{min_cdc_value}'
          AND datasource_num_id IN (3000, 4000, 5000, 6000, 7000, 8000, 9000, 10000, 11000, 12001, 12002, 12003, 12004, 12005, 13000, 14000)
    )
)
WHERE row_num = 1
""")


# COMMAND ----------

spark.sql(f"""
CREATE OR REPLACE TEMPORARY VIEW dim_customer AS
SELECT * FROM (
SELECT *,
row_number() OVER (
PARTITION BY customer_dim_key
ORDER BY updated_ts DESC
) AS row_num
FROM (
SELECT
customer_dim_key, customer_no, customer_name, intercompany_customer,
national_acct_no, dl_load_date, updated_ts
FROM brz_teradata.udm_customer_dim
)
)
WHERE row_num = 1
""")

# COMMAND ----------

spark.sql(f"""
CREATE OR REPLACE TEMPORARY VIEW dim_branch AS
SELECT * FROM (
SELECT *,
row_number() OVER (
PARTITION BY branch_id
ORDER BY updated_ts DESC
) AS row_num
FROM (
SELECT branch_id, sbu, updated_ts FROM brz_teradata.branch_dim
)
)
WHERE row_num = 1
""")

# COMMAND ----------

spark.sql("""
CREATE OR REPLACE TEMPORARY VIEW dim_digital_channel AS
SELECT
    TRIM(ent_type) AS ent_type,
    datasource_num_id,
    TRIM(rollup_category) AS rollup_category
FROM (
    SELECT
        ent_type,
        datasource_num_id,
        rollup_category,
        ROW_NUMBER() OVER (
            PARTITION BY
                TRIM(ent_type),
                datasource_num_id,
                TRIM(rollup_category)
            ORDER BY updated_ts DESC
        ) AS rn
    FROM brz_teradata.udm_digital_channel_code_dim
) t
WHERE rn = 1
""")

# COMMAND ----------

spark.sql("""
CREATE OR REPLACE TEMPORARY VIEW ref_datasource AS
SELECT
    datasource_num_id,
    datasource_description,
    datasource_name
FROM (
    SELECT
        datasource_num_id,
        datasource_description,
        datasource_name,
        ROW_NUMBER() OVER (
            PARTITION BY datasource_num_id
            ORDER BY dl_load_date DESC
        ) AS rn
    FROM brz_teradata.datasource_tbl
) t
WHERE rn = 1
""")
